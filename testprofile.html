БАЗА ДАННЫХ --- DB_PATH = "bot.db" def init_db(): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" CREATE TABLE IF NOT EXISTS users ( user_id INTEGER PRIMARY KEY, username TEXT, current_persona TEXT DEFAULT 'alena', spawn_location TEXT DEFAULT 'club', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) """) cur.execute(""" CREATE TABLE IF NOT EXISTS history ( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, persona TEXT, role TEXT, content TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) """) conn.commit() conn.close() logger.info("✅ База данных инициализирована") def get_user_persona(user_id: int) -> str: conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute("SELECT current_persona FROM users WHERE user_id = ?", (user_id,)) row = cur.fetchone() conn.close() return row[0] if row else "alena" def set_user_persona(user_id: int, persona: str): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" INSERT INTO users (user_id, current_persona) VALUES (?, ?) ON CONFLICT(user_id) DO UPDATE SET current_persona = excluded.current_persona """, (user_id, persona)) conn.commit() conn.close() def get_user_location(user_id: int) -> str: conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute("SELECT spawn_location FROM users WHERE user_id = ?", (user_id,)) row = cur.fetchone() conn.close() return row[0] if row else "club" def set_user_location(user_id: int, location: str): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" INSERT INTO users (user_id, spawn_location) VALUES (?, ?) ON CONFLICT(user_id) DO UPDATE SET spawn_location = excluded.spawn_location """, (user_id, location)) conn.commit() conn.close() def get_history(user_id: int, limit: int = 10): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" SELECT role, content FROM history WHERE user_id = ? ORDER BY created_at DESC LIMIT ? """, (user_id, limit)) rows = cur.fetchall() conn.close() return [{"role": row[0], "content": row[1]} for row in reversed(rows)] def save_message(user_id: int, persona: str, role: str, content: str): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" INSERT INTO history (user_id, persona, role, content) VALUES (?, ?, ?, ?) """, (user_id, persona, role, content)) conn.commit() conn.close() def clear_history(user_id: int): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute("DELETE FROM history WHERE user_id = ?", (user_id,)) conn.commit() conn.close() def register_user(user_id: int, username: str = None): conn = sqlite3.connect(DB_PATH) cur = conn.cursor() cur.execute(""" INSERT OR IGNORE INTO users (user_id, username, current_persona, spawn_location) VALUES (?, ?, 'alena', 'club') """, (user_id, username)) conn.commit() conn.close() init_db() # --- 4.
